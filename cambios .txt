function construirInformeCompleto(data) {
    if (!data) return;
    const r = data.resumen;
    const fecha = new Date(data.fecha_cierre || data.fecha_creacion).toLocaleString('es-VE');

    const filaTabla = function(items, signo) {
        if (!items.length) return '<tr><td colspan="7" class="text-center">Ninguno</td></tr>';
        return items.map(function(f) {
            const dif = parseFloat(f.diferencia);
            return '<tr>' +
                '<td>' + (f.codigo || '') + '</td>' +
                '<td>' + (f.nombre || '') + '</td>' +
                '<td>' + (f.unidad || 'und') + '</td>' +
                '<td class="text-end">' + parseFloat(f.stock_teorico).toFixed(3) + '</td>' +
                '<td class="text-end">' + parseFloat(f.stock_fisico).toFixed(3) + '</td>' +
                '<td class="text-end">' + (signo && dif > 0 ? '+' : '') + dif.toFixed(3) + '</td>' +
                '<td class="text-end">$ ' + parseFloat(f.valor_diferencia_usd).toFixed(2) + '</td>' +
            '</tr>';
        }).join('');
    };

    const tabla = function(titulo, items, signo) {
        return '<h2>' + titulo + ' (' + items.length + ')</h2>' +
            '<table>' +
                '<thead><tr>' +
                    '<th>Código</th><th>Producto</th><th>Unidad</th>' +
                    '<th class="text-end">Stock Teórico</th><th class="text-end">Stock Físico</th>' +
                    '<th class="text-end">Diferencia</th><th class="text-end">Valor USD</th>' +
                '</tr></thead>' +
                '<tbody>' + filaTabla(items, signo) + '</tbody>' +
            '</table>';
    };

    document.getElementById('informe-completo-impresion').innerHTML =
        '<div class="doc-header">' +
            '<h1>INFORME DE TOMA FÍSICA #' + data.toma_id + '</h1>' +
            '<div><b>Almacén:</b> ' + data.almacen + ' &nbsp;|&nbsp; <b>Tipo:</b> ' + data.tipo + ' &nbsp;|&nbsp; <b>Estado:</b> ' + data.estado + '</div>' +
            '<div><b>Fecha:</b> ' + fecha + '</div>' +
        '</div>' +
        '<table class="doc-meta"><tr>' +
            '<td><b>Total líneas:</b> ' + r.total_lineas + '</td>' +
            '<td><b>Cuadrados:</b> ' + r.cuadrados + '</td>' +
            '<td><b>Faltantes:</b> ' + r.faltantes + '</td>' +
            '<td><b>Sobrantes:</b> ' + r.sobrantes + '</td>' +
        '</tr></table>' +
        '<div class="resumen-box">' +
            '<div class="resumen-item"><span class="num" style="color:#b02a37">$ ' + r.valor_faltantes_usd.toFixed(2) + '</span>Valor Faltantes</div>' +
            '<div class="resumen-item"><span class="num" style="color:#146c43">$ ' + r.valor_sobrantes_usd.toFixed(2) + '</span>Valor Sobrantes</div>' +
            '<div class="resumen-item"><span class="num">$ ' + r.valor_neto_usd.toFixed(2) + '</span>Diferencia Neta</div>' +
        '</div>' +
        tabla('PRODUCTOS FALTANTES', data.faltantes, false) +
        tabla('PRODUCTOS SOBRANTES', data.sobrantes, true) +
        tabla('PRODUCTOS CUADRADOS', data.cuadrados, true) +
        '<div class="firma"><div class="linea"></div><div> Firma del Responsable</div></div>';
}
